ErrorDocument 404 /404.html
AddType image/avif .avif
AddType image/webp .webp
AddType font/woff2 .woff2
AddType application/manifest+json .webmanifest

<IfModule mod_brotli.c>
  AddOutputFilterByType BROTLI_COMPRESS text/html text/css text/plain text/xml application/javascript application/json application/manifest+json image/svg+xml
</IfModule>
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css text/plain text/xml application/javascript application/json application/manifest+json image/svg+xml
</IfModule>
<IfModule mod_headers.c>
  <FilesMatch "\.(avif|webp|jpg|jpeg|png|gif|svg|ico|woff2|pdf|mp4|doc)$">
    Header set Cache-Control "max-age=31536000, immutable"
  </FilesMatch>
  <FilesMatch "\.(html|css|js|json|xml|txt|webmanifest)$">
    Header set Cache-Control "max-age=600, must-revalidate"
  </FilesMatch>
  # písma aj pre stránku v ráme bez allow-same-origin (pôvod null, napr. náhľad): bez hlavičky prehliadač písmo odmietne
  <FilesMatch "\.woff2$">
    Header set Access-Control-Allow-Origin "*"
  </FilesMatch>
</IfModule>
